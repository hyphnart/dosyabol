cmake_minimum_required(VERSION 3.21)
project(DosyaBol VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets)
add_executable(DosyaBol WIN32 src/main.cpp src/SplitEngine.cpp src/SplitEngine.h)
target_link_libraries(DosyaBol PRIVATE Qt6::Core Qt6::Widgets)
if(MSVC)
  target_compile_options(DosyaBol PRIVATE /W4 /utf-8)
else()
  target_compile_options(DosyaBol PRIVATE -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(DosyaBolTests tests/split_tests.cpp src/SplitEngine.cpp src/SplitEngine.h)
  target_include_directories(DosyaBolTests PRIVATE src)
  target_link_libraries(DosyaBolTests PRIVATE Qt6::Core)
  add_test(NAME SplitRoundtrip COMMAND DosyaBolTests)
endif()
