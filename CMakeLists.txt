cmake_minimum_required(VERSION 3.21)
project(DosyaBol VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets)

# Bağımsız çözücü: Qt gerektirmez, statik CRT ile derlenir; parçaların başına gömülebilir
# veya çıkış klasörüne kopyalanabilir. DosyaBol.exe ile aynı klasörde bulunmalıdır.
add_executable(DosyaBolCozucu src/stub/cozucu_main.cpp src/common/dbcore.h)
set_property(TARGET DosyaBolCozucu PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(DosyaBol WIN32 src/main.cpp src/SplitEngine.cpp src/SplitEngine.h src/common/dbcore.h)
target_link_libraries(DosyaBol PRIVATE Qt6::Core Qt6::Widgets)
add_dependencies(DosyaBol DosyaBolCozucu)

foreach(target DosyaBol DosyaBolCozucu)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
  add_executable(DosyaBolTests tests/split_tests.cpp src/SplitEngine.cpp src/SplitEngine.h src/common/dbcore.h)
  target_include_directories(DosyaBolTests PRIVATE src)
  target_link_libraries(DosyaBolTests PRIVATE Qt6::Core)
  target_compile_definitions(DosyaBolTests PRIVATE "DOSYABOL_STUB_PATH=\"$<TARGET_FILE:DosyaBolCozucu>\"")
  add_dependencies(DosyaBolTests DosyaBolCozucu)
  if(MSVC)
    target_compile_options(DosyaBolTests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(DosyaBolTests PRIVATE -Wall -Wextra)
  endif()
  add_test(NAME SplitRoundtrip COMMAND DosyaBolTests)
endif()
